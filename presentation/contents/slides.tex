    \begin{enumerate}\setlength{\itemsep}{4pt}

\begin{comparisonframe}{Backend: SDK, Provider, dan Integrasi}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.47\textwidth}
            \textbf{GL-IAM SDK}\par
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item Satu interface bekerja dengan banyak provider melalui SIMI.
                \item Provider yang terdokumentasi: PostgreSQL, Keycloak, dan StackAuth.
                \item Cookbook menyediakan contoh DPoP, API-key hierarchy, SSO, dan agent.
            \end{itemize}
        \end{column}
        \begin{column}{0.47\textwidth}
            \textbf{GL Connectors}\par
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item Pola facade untuk SQL, GitHub, Gmail, Drive, dan layanan eksternal.
                \item Mendukung refactor tool Pamela dari BOSA/MCP ke SDK bersama.
                \item Deployment agent divalidasi melalui Helm dan CI/CD.
            \end{itemize}
        \end{column}
    \end{columns}
\end{comparisonframe}
